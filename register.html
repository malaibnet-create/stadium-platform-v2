// تاريخ اليوم في تمام الساعة 00:00:00 للمقارنة العادلة const today = new Date(); today.setHours(0, 0, 0, 0); let weeklyData = {}; // المسح من الأسفل للأعلى لضمان عدم تأثر أرقام الصفوف عند الحذف for (let i = data.length - 1; i >= 1; i--) { let rawDate = data[i][2]; let bookingDate; // محاولة تحويل التاريخ مهما كان تنسيقه (نصي أو كائن تاريخ) if (rawDate instanceof Date) { bookingDate = new Date(rawDate); } else { // إذا كان التاريخ نصاً بتنسيق dd/mm/yyyy let parts = String(rawDate).split('/'); if (parts.length === 3) { bookingDate = new Date(parts[2], parts[1] - 1, parts[0]); } } // إذا كان التاريخ صالحاً وأقل من تاريخ اليوم if (bookingDate && !isNaN(bookingDate.getTime()) && bookingDate < today) { let sId = data[i][0]; // ID الملعب let month = bookingDate.getMonth() + 1; // استخراج الساعة بشكل آمن (يتعامل مع "20:00" أو الرقم 20) let rawHour = data[i][3]; let hour = parseInt(String(rawHour).split(':')[0]); // جلب الأسعار للملعب من ورقة الملاعب let pDay = 0, pNight = 0; for (let j = 1; j < stadiumData.length; j++) { if (stadiumData[j][0] == sId) { pDay = parseFloat(stadiumData[j][8]) || 0; // العمود I pNight = parseFloat(stadiumData[j][9]) || 0; // العمود J break; } } // حساب الدخل (ليلي من 18 مساءً أو قبل 7 صباحاً) let income = (hour >= 18 || hour < 7) ? pNight : pDay; // تجميع البيانات في الذاكرة لتحديث Stats_Summary لاحقاً let key = sId + "_" + month; if (!weeklyData[key]) weeklyData[key] = { hours: 0, income: 0 }; weeklyData[key].hours += 1; weeklyData[key].income += income; // 1. النقل للأرشيف archiveSheet.appendRow(data[i]); // 2. الحذف من صفحة الحجوزات الأصلية bookingSheet.deleteRow(i + 1); Logger.log("تمت أرشفة حجز الملعب: " + sId + " بتاريخ: " + rawDate); } } // 3. تحديث ورقة Stats_Summary بالأرقام المجمعة if (Object.keys(weeklyData).length > 0) { updatePermanentStats(weeklyData, statsSheet); } } /** * تحديث الإحصائيات الدائمة في ورقة Stats_Summary */ function updatePermanentStats(weeklyData, statsSheet) { let sheetData = statsSheet.getDataRange().getValues(); for (let key in weeklyData) { let parts = key.split("_"); let sId = parts[0]; let month = parts[1]; let found = false; // البحث عن سطر موجود لنفس الملعب ونفس الشهر for (let row = 1; row < sheetData.length; row++) { if (sheetData[row][0] == sId && sheetData[row][1] == month) { let currentHours = parseFloat(sheetData[row][2]) || 0; let currentIncome = parseFloat(sheetData[row][3]) || 0; // تحديث القيم (القديم + الجديد) statsSheet.getRange(row + 1, 3).setValue(currentHours + weeklyData[key].hours); statsSheet.getRange(row + 1, 4).setValue(currentIncome + weeklyData[key].income); found = true; break; } } // إذا لم يوجد سطر سابق، ننشئ سطراً جديداً if (!found) { statsSheet.appendRow([sId, month, weeklyData[key].hours, weeklyData[key].income]); } } }
